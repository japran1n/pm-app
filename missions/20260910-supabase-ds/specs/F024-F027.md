## F024 — Install and wire next-themes

### Step 1: Install
```bash
npm install next-themes@0.4.6
```

### Step 2: Create ThemeProvider wrapper
Create `components/theme-provider.tsx`:
```tsx
"use client";

import { ThemeProvider as NextThemesProvider } from "next-themes";
import type { ThemeProviderProps } from "next-themes";

export function ThemeProvider({ children, ...props }: ThemeProviderProps) {
  return <NextThemesProvider {...props}>{children}</NextThemesProvider>;
}
```

### Step 3: Update app/layout.tsx
Wrap the body content with ThemeProvider. The key config:
```tsx
import { ThemeProvider } from "@/components/theme-provider";

// Inside RootLayout, wrap children:
<ThemeProvider
  attribute={["class", "data-theme"]}
  defaultTheme="dark"
  enableSystem
  disableTransitionOnChange
>
  {children}
  <Toaster />
</ThemeProvider>
```

Also: REMOVE `dark` className and `data-theme="dark"` from the `<html>` element (next-themes manages these now).
Keep only `{inter.variable} ${sourceCodePro.variable} h-full antialiased` on `<html>`.

### Step 4: No-flash script
Add to `<head>` in layout.tsx a script that sets dark mode before React hydrates:
```tsx
<script
  dangerouslySetInnerHTML={{
    __html: `(function(){try{var t=localStorage.getItem('theme');var d=document.documentElement;if(t==='light'){d.classList.remove('dark');d.removeAttribute('data-theme')}else{d.classList.add('dark');d.setAttribute('data-theme','dark')}}catch(e){}})();`,
  }}
/>
```
This prevents flash of wrong theme on page load.

## F025 — Theme switch control in workspace UI

Create `components/ui/theme-toggle.tsx`:
```tsx
"use client";

import { Moon, Sun } from "lucide-react";
import { useTheme } from "next-themes";
import { Button } from "@/components/ui/button";

export function ThemeToggle() {
  const { theme, setTheme } = useTheme();

  return (
    <Button
      variant="ghost"
      size="icon-sm"
      onClick={() => setTheme(theme === "dark" ? "light" : "dark")}
      aria-label="Toggle theme"
    >
      <Sun className="h-4 w-4 rotate-0 scale-100 transition-all dark:-rotate-90 dark:scale-0" />
      <Moon className="absolute h-4 w-4 rotate-90 scale-0 transition-all dark:rotate-0 dark:scale-100" />
    </Button>
  );
}
```
